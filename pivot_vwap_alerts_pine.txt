// Function to calculate pivot levels
calcLevels(high1, low1, close1) =>
    pivot = (high1 + low1 + close1) / 3
    BC = (high1 + low1) / 2
    TC = pivot - BC + pivot
    r1 = 2 * pivot - low1
    s1 = 2 * pivot - high1
    r2 = pivot + (high1 - low1)
    s2 = pivot - (high1 - low1)
    r3 = high1 + 2 * (pivot - low1)
    s3 = low1 - 2 * (high1 - pivot)
    //r4 = high1 + 3 * (pivot - low1)
    //s4 = low1 - 3 * (high1 - pivot)
    r4 = r3 + r2 - r1
    s4 = s3 + s2 - s1
    [pivot, BC, TC, r1, s1, r2, s2, r3, s3, r4, s4]

drawHorizontalLine(x1, x2, y, clr, draw) =>
    if(draw)
        l = line.new(x1, y, x2, y, xloc.bar_time, color = clr)
        line.delete(l[1])

drawLabel(x, y, txt, clr, draw=true) =>
    if draw
        l = label.new(x, y, text = txt, xloc = xloc.bar_time, textcolor = clr, style = label.style_none)
        label.delete(l[1])
        drawHorizontalLine(x, x+86400000, y, clr, draw)

calculateLongShortConditions(level) =>
    // Store individual crossover results in variables
    [O_t, H_t, L_t, C_t] = request.security(syminfo.tickerid, strategyTF, [open, high, low, close], lookahead = barmerge.lookahead_on)
    highCrossOver = ta.crossover(H_t, level)
    closeCrossOver = ta.crossover(C_t, level)
    openCrossOver = ta.crossover(O_t, level)
    lowCrossOver = ta.crossover(L_t, level)

    highCrossUnder = ta.crossunder(H_t, level)
    closeCrossUnder = ta.crossunder(C_t, level)
    openCrossUnder = ta.crossunder(O_t, level)
    lowCrossUnder = ta.crossunder(L_t, level)

    // Combine variables to build the final long and short conditions
    longCondition = highCrossOver or closeCrossOver or openCrossOver or lowCrossOver
    shortCondition = highCrossUnder or closeCrossUnder or openCrossUnder or lowCrossUnder
    [longCondition, shortCondition]

// Function to raise alerts
raiseAlert(level, label) =>
    [enterLong, enterShort] = calculateLongShortConditions(level)
    if enterLong
        alert("Price crossed above "  + str.tostring(level, "0.#") + ": " + label, alert.freq_once_per_bar)
    else if enterShort
        alert("Price crossed below "  + str.tostring(level, "0.#") + ": " + label, alert.freq_once_per_bar)
    //else 
    //    log.info("No alert for now...")
    [enterLong, enterShort]

// --- Find the next higher timeframe automatically ---
// This function returns the next highest timeframe string from a predefined list.
f_getHigherTimeframe(current_tf) =>
    higher_tf = ""
    // Pre-defined ordered list of common timeframes.
    var timeframes = array.from("1", "3", "5", "15", "30", "60", "120", "180", "240", "D", "W", "M")
    
    // Find the index of the current timeframe in the list.
    index = array.indexof(timeframes, current_tf)
    
    // If the current timeframe is found and is not the last in the list,
    // get the next one.
    if index != -1 and index < array.size(timeframes) - 1
        higher_tf := array.get(timeframes, index + 1)
        
    higher_tf
